337template<typename Class, typename ConstructorScheme, typename Bases>
338sol::usertype<Class> new_usertype(
339 sol::state_view &lua,
340 Bases &&bases,
341 ConstructorScheme &&constructor
342)
343{
344 static_assert( detail::luna_traits<Class>::impl, "Type must implement luna_traits<T>" );
345
346 // Ensure global doctable exists
347 sol::table global_dt = detail::get_global_doctable( lua );
348
349 constexpr std::string_view name = detail::luna_traits<Class>::name;
350
351 // Register Sol usertype
352 sol::usertype<Class> ut;
353 using BasesBare = detail::remove_cv_ref_t<Bases>;
354 if constexpr( std::is_same_v<BasesBare, no_bases_t> ) {
355 ut = lua.new_usertype<Class>( name, constructor );
356 } else {
357 ut = lua.new_usertype<Class>( name, constructor, sol::base_classes, bases );
358 }
359
360 // Create doctable for this type
361 sol::table type_dt = lua.create_table();
362 global_dt[detail::KEY_TYPES][name] = type_dt;
363
364 // Document Sol usertype
365 type_dt[detail::KEY_TYPE_IMPL] = ut;
366
367 // Init members table
368 type_dt[detail::KEY_MEMBER] = std::vector<sol::object>();
369
370 // Document constructors (or lack thereof)
371 detail::doc_constructors( type_dt, constructor );
372
373 // Document bases (or lack thereof)
374 detail::doc_bases( type_dt, bases );
375
376 // Add helper method to get name under which usertype is bound in Lua
377 ut[detail::KEY_GET_TYPE] = []() -> std::string_view {
378 return detail::luna_traits<Class>::name;
379 };
380
381 return ut;
382}
